    private suspend fun pingNodeListPerNode(targets: List<NodeEntity>, generation: Long) {
        val cfg = _settings.value
        val limit = cfg.pingConcurrency.coerceIn(1, adaptivePingConcurrency)
        val targetIds = targets.map { it.id }
        val pending = java.util.Collections.synchronizedSet(targetIds.toMutableSet())
        val completedPings = java.util.Collections.synchronizedMap(mutableMapOf<String, Int>())
        val semaphore = kotlinx.coroutines.sync.Semaphore(limit)
        // coroutineScope gives this suspend fun the CoroutineScope receiver
        // `async` needs; the scope ends when every probe settles, so the
        // structured-concurrency guarantee from the previous inline version
        // (all children awaited before the caller continues) is preserved.
        val finished = kotlinx.coroutines.coroutineScope {
            val jobs = targets.map { node ->
                async {
                    try {
                        semaphore.withPermit {
                            // Real tunnel probe: spawn sing-box with this
                            // node's config, wait for the endpoint to come
                            // up, then measure an HTTP round trip through
                            // the tunnel. A node that answers a bare TCP
                            // connect but whose handshake fails must NOT
                            // read a latency here — the number on the row
                            // has to reflect whether the tunnel actually
                            // works, which is what the user sees in the
                            // Smart pick and what V2rayNG shows.
                            val ping = performRealPing(node)
                            val persistedPing = ping.coerceAtLeast(0)
                            nodeDao.updatePing(node.id, persistedPing)
                            completedPings[node.id] = persistedPing
                        }
                    } catch (e: kotlinx.coroutines.CancellationException) {
                        throw e
                    } catch (e: Exception) {
                        runCatching {
                            nodeDao.updatePing(node.id, 0)
                            completedPings[node.id] = 0
                        }
                        log("Ping failed for ${node.name}: ${e.message}")
                    } finally {
                        synchronized(pending) {
                            pending.remove(node.id)
                            if (bulkPingGeneration == generation) {
                                _pingingNodeIds.value = pending.toSet()
                            }
                        }
                    }
                }
            }
            withTimeoutOrNull(PingBudget.batchMs(targets.size, limit)) {
                jobs.awaitAll()
            } != null
        }
        if (!finished) {
            val leftovers = synchronized(pending) { pending.toList() }
            runCatching { nodeDao.updatePingsBatch(leftovers.map { Pair(it, 0 as Int?) }) }
            log("Ping deadline reached: ${leftovers.size} node(s) marked unreachable")
        }
        if (finished && cfg.pingAutoDeleteUnreachable) {
            val removed = removeUnhealthyNodesAfterPing(
                completedPings = completedPings.toMap(),
                thresholdMs = cfg.pingAutoDeleteThresholdMs
            )
            if (removed > 0) log("Auto-removed $removed node(s) after the ping test")
        }
    }

    private fun pingNodeListInternal(targets: List<NodeEntity>) {
        if (targets.isEmpty()) return
        val generation = ++bulkPingGeneration
        bulkPingJob?.cancel()
        val job = viewModelScope.launch(
            Dispatchers.IO,
            start = kotlinx.coroutines.CoroutineStart.LAZY
        ) {
            _isPinging.value = true
            _smartTesting.value = true
            try {
                val targetIds = targets.map { it.id }
                _serverTestResults.value = _serverTestResults.value - targetIds.toSet()
                _pingingNodeIds.value = targetIds.toSet()
                // Previous ping values are intentionally left in Room
                // while the new pass runs. The old reset-to-null made
                // every cold start show "—" on every row for the 6-8 s
                // the pass took, even though the previous numbers were
                // already in the database and still a reasonable
                // estimate. The spinner on each row (fed by
                // _pingingNodeIds) marks the row as under measurement;
                // the number updates the moment a fresh sample lands.
                // A ping pass that fails outright still ends with the
                // deadline path marking its leftovers as 0, so no stale
                // number outlives a genuinely unreachable node.

                // Real tunnel probe for every protocol, in the fastest
                // shape each one allows. urltest runs first: one sing-box
                // holds all the TCP-outbound nodes (VLESS/VMess/Trojan/SS)
                // and answers their Clash API /delay in parallel, so a
                // dozen rows land in a few seconds. The UDP-only nodes
                // (OVPN/WG/Hysteria2) cannot be expressed as raw outbounds
                // and are measured on the per-node path, where the
                // normalizer turns each one into a valid endpoint and the
                // full tunnel probe decides whether the handshake actually
                // completes. The user's ping-type setting no longer
                // decides the shape — the protocol does, because only a
                // tunnel round trip answers the question the row is meant
                // to answer: does this config actually work.
                val viaUrlTest = runCatching {
                    pingNodeListViaUrlTest(targets)
                }.getOrElse { e ->
                    log("urltest bulk ping crashed: ${e.javaClass.simpleName}: ${e.message}")
                    false
                }
                if (!viaUrlTest) {
                    log("urltest produced no result; running per-node fallback")
                    pingNodeListPerNode(targets, generation)
                }
                log("Ping finished for ${targets.size} node(s)")
                runCatching {
                    // TEMP DEBUG: see NodeDao.debugFindByNamePattern. Compare
                    // this list's size and ids against what the UI shows and
                    // against the "xray measured X/Y" ids logged above -
                    // that comparison alone tells us duplicate-DB-rows vs
                    // UI-cache with no ambiguity left.
                    val dbRows = nodeDao.debugFindByNamePattern("%xhttp%")
                    log(
                        "DEBUG xhttp-name rows in DB: ${dbRows.size} -> " +
                            dbRows.joinToString(" | ") { "${it.id.take(8)}:ping=${it.pingMs}" }
                    )
                }.onFailure { log("DEBUG dump failed: ${it.javaClass.simpleName}: ${it.message}") }
        lastAutoPingCompletedAt = System.currentTimeMillis()
        lastAutoPingIds = targets.map { it.id }.toSet()
            } finally {
                if (bulkPingGeneration == generation) {
                    bulkPingJob = null
                    _pingingNodeIds.value = emptySet()
                    _isPinging.value = false
                    _smartTesting.value = false
                }
            }
        }
        bulkPingJob = job
        job.start()
    }
